import ctypes,sys,time
x=ctypes.CDLL('libX11.so.6');t=ctypes.CDLL('libXtst.so.6');p=ctypes.c_void_p
x.XOpenDisplay.argtypes=[ctypes.c_char_p];x.XOpenDisplay.restype=p
x.XFlush.argtypes=[p];x.XCloseDisplay.argtypes=[p]
x.XStringToKeysym.argtypes=[ctypes.c_char_p];x.XStringToKeysym.restype=ctypes.c_ulong
x.XKeysymToKeycode.argtypes=[p,ctypes.c_ulong];x.XKeysymToKeycode.restype=ctypes.c_ubyte
t.XTestFakeMotionEvent.argtypes=[p,ctypes.c_int,ctypes.c_int,ctypes.c_int,ctypes.c_ulong]
t.XTestFakeButtonEvent.argtypes=[p,ctypes.c_uint,ctypes.c_int,ctypes.c_ulong]
t.XTestFakeKeyEvent.argtypes=[p,ctypes.c_uint,ctypes.c_int,ctypes.c_ulong]
d=x.XOpenDisplay(b':97')
if not d:raise RuntimeError('No emulator X display')
def key(k,down):
 code=x.XKeysymToKeycode(d,x.XStringToKeysym(k.encode()));t.XTestFakeKeyEvent(d,code,down,0)
t.XTestFakeButtonEvent(d,1,0,0);x.XFlush(d);time.sleep(.5)
if sys.argv[1]=='click':
 t.XTestFakeMotionEvent(d,-1,int(sys.argv[2]),int(sys.argv[3]),0);x.XFlush(d);time.sleep(.5)
 for i in range(int(sys.argv[4]) if len(sys.argv)>4 else 1):
  t.XTestFakeButtonEvent(d,1,1,0);x.XFlush(d);time.sleep(.5);t.XTestFakeButtonEvent(d,1,0,0);x.XFlush(d);time.sleep(.5)
elif sys.argv[1]=='drag':
 t.XTestFakeMotionEvent(d,-1,int(sys.argv[2]),int(sys.argv[3]),0);x.XFlush(d);time.sleep(.5);t.XTestFakeButtonEvent(d,1,1,0);x.XFlush(d);time.sleep(.3);
 for n in range(1,21):
  px=int(sys.argv[2])+(int(sys.argv[4])-int(sys.argv[2]))*n//20;py=int(sys.argv[3])+(int(sys.argv[5])-int(sys.argv[3]))*n//20
  t.XTestFakeMotionEvent(d,-1,px,py,0);x.XFlush(d);time.sleep(.1)
 time.sleep(1);t.XTestFakeButtonEvent(d,1,0,0);x.XFlush(d);time.sleep(.5)
elif sys.argv[1]=='key':
 ks=sys.argv[2].split('+')
 for k in ks:key(k,1)
 x.XFlush(d);time.sleep(.12)
 for k in reversed(ks):key(k,0)
 x.XFlush(d);time.sleep(.6)
elif sys.argv[1]=='type':
 for k in sys.argv[2]:
  if k.isupper() or k==':':key('Shift_L',1)
  name={' ':'space',':':'colon','/':'slash','.':'period','-':'minus'}.get(k,k.lower())
  key(name,1);x.XFlush(d);time.sleep(.1);key(name,0)
  if k.isupper() or k==':':key('Shift_L',0)
  x.XFlush(d);time.sleep(.025)
x.XFlush(d);x.XCloseDisplay(d)
