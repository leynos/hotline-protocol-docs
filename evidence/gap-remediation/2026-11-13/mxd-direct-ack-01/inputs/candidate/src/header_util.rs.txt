//! Helpers for working with frame headers.
//!
//! Provides utility functions for building direct reply
//! [`crate::transaction::FrameHeader`] values. These helpers keep the framing
//! logic centralized and consistent across commands.

/// Build a direct reply `FrameHeader` with the request ID, payload size and error code.
/// Direct replies use wire type zero; the request type remains with the caller.
///
/// # Panics
/// Panics if `payload_len` does not fit within `u32`.
///
/// # Aborts
/// Under Kani verification builds, aborts if `payload_len` does not fit within
/// `u32`.
#[must_use]
pub fn reply_header(
    req: &crate::transaction::FrameHeader,
    payload_error: u32,
    payload_len: usize,
) -> crate::transaction::FrameHeader {
    let Ok(payload_len_u32) = u32::try_from(payload_len) else {
        payload_len_overflow(payload_len);
    };
    crate::transaction::FrameHeader {
        flags: 0,
        is_reply: 1,
        ty: 0,
        id: req.id,
        error: payload_error,
        total_size: payload_len_u32,
        data_size: payload_len_u32,
    }
}

#[cfg(kani)]
#[cold]
fn payload_len_overflow(_payload_len: usize) -> ! { std::process::abort() }

#[cfg(not(kani))]
#[cold]
fn payload_len_overflow(payload_len: usize) -> ! {
    panic!("payload length exceeds u32::MAX: {payload_len}");
}

#[cfg(kani)]
mod kani;
