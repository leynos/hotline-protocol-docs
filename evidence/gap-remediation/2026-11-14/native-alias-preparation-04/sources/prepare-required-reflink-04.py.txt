"""Prepare only the stopped R04 alias lab; never boot or contact the server."""
import hashlib

from datetime import datetime, timezone
from pathlib import Path

ROOT = Path("/data/leynos/Projects/mothra")
P = ROOT / "state/native-sessions/native-alias-live-preparation-04"
S = P / "stopped-preparation"
R = ROOT / "state/mac-runtime/native-alias-live-04"
BASE = ROOT / "state/mac-runtime/native-board-02/base.dsk"
ROM = ROOT / "state/mac-runtime/native-board-02/rom"
THRESHOLD = 10_536_091_648
EXPECTED = {
    BASE: (8_388_608_000, 0o444, 1678261666, "e5cbbaaf28481bafbfa6dfe87dd1cac20d71624fe2fbd79c15373a365bde9910"),
    ROM: (1_945_746, 0o644, 1678261665, "18da89e42c7cb68bffe0ec8e404ddd6b2527947d8d5fb6f694a49b2d4a89d2bd"),
}
FORKS = {
    "admin": ("11193", 734, "26612ad46cd91b90050feacf517d32a9e56d4a4aa92fbe6618eaf6e9e5ac60a3"),
    "fixture-low": ("11201", 734, "fce81359d0efcfc6c2dbf42852db66b0fb7a404b2889162b8f8fad81361ce2d5"),
    "Prefs": ("11202", 175, "4ed2b3e14e6e8a6cf246d16f7d1b98f0b91d1b09f6ab5497de1056a6f617d196"),
    "server-data": ("11183-4352-0", 616821, "ad7040db4eb67aef1308d915d80305dc96b4aa437a5b3654ea9bb76ab2e4cd1b"),
    "server-rsrc": ("11183-4353-1", 38460, "fe83a2cfaacd80e998a8ca274a828a0196d5cd5c4db59b06c1763cb3c1c48b12"),
}
TSK = ROOT / "tools/sleuthkit/usr/bin"
TSK_ENV = {**os.environ, "LD_LIBRARY_PATH": str(ROOT / "tools/sleuthkit/usr/lib64")}
commands = []
def command(label, argv, *, expected=0, env=None, save_bytes=False):
    result = subprocess.run(argv, capture_output=True, env=env, check=False)
    row = {"label": label, "argv": list(map(str, argv)), "exit": result.returncode,
           "stdout_sha256": hashlib.sha256(result.stdout).hexdigest(),
           "stdout_size": len(result.stdout), "stderr_sha256": hashlib.sha256(result.stderr).hexdigest(),
           "stderr_size": len(result.stderr)}
    commands.append(row)
    (S / "commands.json").write_text(json.dumps(commands, indent=2) + "\n")
    if not label.startswith("icat-"):
        (S / f"{label}.stdout").write_bytes(result.stdout)
    (S / f"{label}.stderr").write_bytes(result.stderr)
    if result.returncode != expected:
        raise RuntimeError(f"{label}: exit {result.returncode}, expected {expected}")
    return result.stdout
def regular(path, *, size=None, mode=None):
    if path.is_symlink():
        raise ValueError(f"symlink: {path}")
    info = path.stat()
    if not stat.S_ISREG(info.st_mode) or info.st_nlink != 1:
        raise ValueError(f"not a sole regular file: {path}")
    if size is not None and info.st_size != size:
        raise ValueError(f"wrong size: {path}")
    if mode is not None and stat.S_IMODE(info.st_mode) != mode:
        raise ValueError(f"wrong mode: {path}")
    return info
def free(path):
    data = command(f"fuser-{len(commands):03d}", ["sudo", "-n", "fuser", str(path)], expected=1)
    if data or commands[-1]["stderr_size"]:
        raise RuntimeError(f"fuser returned output for {path}")
def digest(path):
    free(path)  # Every image/ROM byte read has its own positive FD-free gate.
    with path.open("rb") as stream:
        value = hashlib.file_digest(stream, "sha256").hexdigest()
    return value


def identity(path):
    info = regular(path)
    return {"path": str(path), "dev": info.st_dev, "inode": info.st_ino,
            "size": info.st_size, "mode": oct(stat.S_IMODE(info.st_mode)),
            "mtime_ns": info.st_mtime_ns, "ctime_ns": info.st_ctime_ns,
            "atime_ns": info.st_atime_ns, "nlink": info.st_nlink,
            "uid": info.st_uid, "gid": info.st_gid, "sha256": digest(path)}


def capacity():
    available = os.statvfs(R.parent).f_bavail * os.statvfs(R.parent).f_frsize
    if available < THRESHOLD:
        raise RuntimeError(f"free {available} < required {THRESHOLD}")
    return available


def clone(label, source, target):
    if target.exists() or target.is_symlink():
        raise FileExistsError(target)
    free(source)
    capacity()
    command(label, ["cp", "--reflink=always", "--", str(source), str(target)])


def inspect_hfs(working):
    free(working)
    full = command("catalog-full", [str(TSK / "fls"), "-r", "-p", "-f", "hfs", str(working)],
                   env=TSK_ENV, save_bytes=True)
    free(working)
    allocated = command("catalog-allocated", [str(TSK / "fls"), "-u", "-r", "-p", "-f", "hfs", str(working)],
                        env=TSK_ENV, save_bytes=True)
    text = allocated.decode(errors="replace")
    users = [line for line in text.splitlines() if "\tHotline19/Users/" in line and line.endswith("/UserData")]
    if users != ["-/r 11193:\tHotline19/Users/admin/UserData", "-/r 11201:\tHotline19/Users/fixture-low/UserData"]:
        raise ValueError("original two-user HFS baseline failed")
    files_news = [line for line in text.splitlines() if line.endswith(("Hotline19/Files", "Hotline19/News"))]
    if files_news != ["d/d 11189:\tHotline19/Files", "d/d 11190:\tHotline19/News"]:
        raise ValueError("Files/News catalogue mismatch")
    if "MessageBoard.txt" not in text or "Banned" in text or "m-alias54-" in text:
        raise ValueError("board, ban or owned-marker catalogue mismatch")
    forks = {}
    for name, (catalog, size, expected_hash) in FORKS.items():
        free(working)
        data = command(f"icat-{name}", [str(TSK / "icat"), "-f", "hfs", str(working), catalog], env=TSK_ENV)
        actual = hashlib.sha256(data).hexdigest()
        if len(data) != size or actual != expected_hash:
            raise ValueError(f"fork mismatch: {name}")
        forks[name] = {"catalog": catalog, "size": len(data), "sha256": actual}
    free(working)
    metadata = command("istat-board", [str(TSK / "istat"), "-f", "hfs", str(working), "11191"], env=TSK_ENV, save_bytes=True)
    free(working)
    board = command("icat-board", [str(TSK / "icat"), "-f", "hfs", str(working), "11191"], env=TSK_ENV)
    for needle in (b"File Path: /Hotline19/MessageBoard.txt", b"Size:\t0", b"File type:\t54455854  TEXT",
                   b"File creator:\t74747874  ttxt", b"Resource fork size:\t0"):
        if needle not in metadata:
            raise ValueError(f"board baseline mismatch: {needle!r}")
    if board:
        raise ValueError("board DATA is not empty")
    return ({"full_lines": len(full.splitlines()), "full_sha256": hashlib.sha256(full).hexdigest(),
             "allocated_lines": len(allocated.splitlines()), "allocated_sha256": hashlib.sha256(allocated).hexdigest(),
             "active_users": users, "files_news": files_news,
             "banned_absent": True}, forks,
            {"catalog": 11191, "data_size": 0, "resource_size": 0, "finder_type": "TEXT", "finder_creator": "ttxt"})


def stage_target():
    shared = R / "shared"
    target = shared / "m-alias54-target"
    side_dir = shared / ".finf"
    side = side_dir / target.name
    payload = b"alias54-live-target\n"
    finder = b"U54TM54X" + bytes(24)
    side_dir.mkdir(mode=0o700)
    for path, data in ((target, payload), (side, finder)):
        with path.open("xb") as stream:
            stream.write(data)
            stream.flush()
            os.fsync(stream.fileno())
        path.chmod(0o600)
        if stat.S_IMODE(path.stat().st_mode) != 0o600:
            raise ValueError(f"staged mode mismatch: {path}")
        if digest(path) != hashlib.sha256(data).hexdigest():
            raise ValueError(f"staged Finder file mismatch: {path}")
    if stat.S_IMODE(side_dir.stat().st_mode) != 0o700:
        raise ValueError("Finder metadata directory mode mismatch")
    def receipt(path):
        value = identity(path)
        value["device"] = value["dev"]
        return value
    return {"shared_parent": identity_dir(shared),
            "target_leaf": receipt(target), "finf": {"status": "stopped_finf_staged",
            "target_before": receipt(target), "target_after": receipt(target),
            "sidecar": receipt(side) | {"first_eight_hex": finder[:8].hex(),
                                      "trailing_24_zero": True, "full_hex": finder.hex()},
            "first_eight_hex": finder[:8].hex(), "trailing_24_zero": True, "full_hex": finder.hex(),
            "finf_dir": identity_dir(side_dir)}}


def identity_dir(path):
    info = path.lstat()
    if not stat.S_ISDIR(info.st_mode) or path.is_symlink():
        raise ValueError(f"not a real directory: {path}")
    return {"path": str(path), "dev": info.st_dev, "inode": info.st_ino, "mode": oct(stat.S_IMODE(info.st_mode)),
            "uid": info.st_uid, "gid": info.st_gid, "nlink": info.st_nlink, "size": info.st_size,
            "mtime_ns": info.st_mtime_ns}


def main():
    if (os.geteuid(), os.getegid()) != (1000, 1000):
        raise PermissionError("stopped preparation requires UID/GID 1000")
    for path in (R, S, ROOT / "state/mac-displays/native-alias-live-04",
                 ROOT / "state/native-sessions/native-alias-live-04"):
        if path.exists() or path.is_symlink():
            raise FileExistsError(path)
    S.mkdir(mode=0o700)
    report = {"status": "started", "started_utc": datetime.now(timezone.utc).isoformat(),
              "method": "required_reflink_both_disks", "commands": commands}
    try:
        ticks_before = {pid: Path(f"/proc/{pid}/stat").read_text().split(") ")[1].split()[19]
                        for pid in (1099659, 1012834)}
        if ticks_before != {1099659: "767842", 1012834: "2357813"}:
            raise ValueError("original process identity changed")
        report["original_ticks_before"] = ticks_before
        before = {str(path): identity(path) for path in EXPECTED}
        for path, (size, mode, inode, sha) in EXPECTED.items():
            if (before[str(path)]["size"], int(before[str(path)]["mode"], 8), before[str(path)]["inode"], before[str(path)]["dev"],
                    before[str(path)]["uid"], before[str(path)]["gid"], before[str(path)]["sha256"]) != (size, mode, inode, 2097, 1000, 1000, sha):
                raise ValueError(f"pinned source mismatch: {path}")
        if BASE.stat().st_dev != R.parent.stat().st_dev or ROM.stat().st_dev != R.parent.stat().st_dev:
            raise ValueError("source and destination must share reviewed filesystem")
        report["source_before"] = before
        report["free_bytes_before"] = capacity()
        R.mkdir(mode=0o700)
        clone("clone-base", BASE, R / "base.dsk")
        clone("clone-working", BASE, R / "working.dsk")
        if (R / "rom").exists() or (R / "rom").is_symlink():
            raise FileExistsError(R / "rom")
        free(ROM)
        command("copy-rom", ["cp", "--", str(ROM), str(R / "rom")])
        (R / "base.dsk").chmod(0o444)
        (R / "working.dsk").chmod(0o644)
        (R / "shared").mkdir(mode=0o755)
        if stat.S_IMODE((R / "shared").stat().st_mode) != 0o755:
            raise ValueError("shared directory mode mismatch")
        prepared = {name: identity(R / name) for name in ("base.dsk", "working.dsk", "rom")}
        for name, mode in (("base.dsk", "0o444"), ("working.dsk", "0o644"), ("rom", "0o644")):
            if (prepared[name]["uid"], prepared[name]["gid"], prepared[name]["mode"], prepared[name]["size"]) != (1000, 1000, mode, EXPECTED[BASE if name != "rom" else ROM][0]):
                raise ValueError(f"prepared owner/mode/size mismatch: {name}")
        if len({before[str(BASE)]["inode"], prepared["base.dsk"]["inode"], prepared["working.dsk"]["inode"]}) != 3:
            raise ValueError("copy inodes overlap")
        if any(prepared[name]["sha256"] != before[str(BASE)]["sha256"] for name in ("base.dsk", "working.dsk")):
            raise ValueError("reflink content mismatch")
        if prepared["rom"]["sha256"] != before[str(ROM)]["sha256"]:
            raise ValueError("ROM content mismatch")
        report["prepared"] = prepared
        report["catalog"], report["forks"], report["board"] = inspect_hfs(R / "working.dsk")
        staged = stage_target()
        for entry in (staged["shared_parent"], staged["target_leaf"], staged["finf"]["sidecar"], staged["finf"]["finf_dir"]):
            if (entry["uid"], entry["gid"]) != (1000, 1000):
                raise ValueError("staged owner mismatch")
        if (staged["target_leaf"]["size"], staged["finf"]["sidecar"]["size"]) != (20, 32):
            raise ValueError("staged size mismatch")
        report["shared_parent"] = staged["shared_parent"]
        report["target_leaf"] = staged["target_leaf"]
        (S / "finf-supplement.json").write_text(json.dumps(staged["finf"], indent=2) + "\n")
        report["source_after"] = {str(path): identity(path) for path in EXPECTED}
        report["original_ticks_after"] = {pid: Path(f"/proc/{pid}/stat").read_text().split(") ")[1].split()[19]
                                          for pid in ticks_before}
        if report["original_ticks_after"] != ticks_before:
            raise ValueError("original process changed during preparation")
        for path in EXPECTED:
            for key in ("dev", "inode", "size", "mode", "mtime_ns", "ctime_ns", "nlink", "sha256"):
                if before[str(path)][key] != report["source_after"][str(path)][key]:
                    raise ValueError(f"source changed: {path} {key}")
        report["free_bytes_at_prep_end"] = capacity()
        (R / "assets.json").write_text(json.dumps({"rom_sha256": prepared["rom"]["sha256"],
            "base_disk_sha256": prepared["base.dsk"]["sha256"], "source_rom": str(ROM),
            "source_disk": str(BASE), "archive_member": None}, indent=2) + "\n")
        report["status"] = "prepared_stopped"
    except BaseException as error:
        report["status"] = "failed"
        report["error"] = f"{type(error).__name__}: {error}"
        raise
    finally:
        (S / "preflight.json").write_text(json.dumps(report, indent=2) + "\n")


if __name__ == "__main__":
    main()
