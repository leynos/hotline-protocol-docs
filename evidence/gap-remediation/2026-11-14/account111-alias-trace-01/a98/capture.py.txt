import asyncio
import hashlib
import json
import os
from datetime import datetime, timedelta, timezone
from pathlib import Path

from mcp import ClientSession, StdioServerParameters
from mcp.client.stdio import stdio_client

ROOT = Path('/data/leynos/Projects/mothra')
OUT = ROOT / 'state/delegation/account111-alias-oracle-02'
BINARY = ROOT / 'evidence/hotline-server-1.9/inputs/HLserver19.pef'
JAR = ROOT / 'tools/ghidra_12.2_PUBLIC/Ghidra/Extensions/GhidraMCP/lib/GhidraMCP.jar'
DATA_SHA = 'ad7040db4eb67aef1308d915d80305dc96b4aa437a5b3654ea9bb76ab2e4cd1b'
JAR_SHA = '201d18ffde71a83a678eae607de265861f37e5a00b989d91958c1197664d2b43'
PROJECT = ROOT / 'state/delegation/endpoint-oracle-01/project'
assert hashlib.sha256(BINARY.read_bytes()).hexdigest() == DATA_SHA
assert hashlib.sha256(JAR.read_bytes()).hexdigest() == JAR_SHA
assert (PROJECT / 'independent-endpoint-01.gpr').is_file()
for name in ('raw', 'config', 'cache', 'data', 'tmp', 'home'):
    (OUT / name).mkdir(parents=True, exist_ok=True)
env = dict(os.environ)
env.update(
    JAVA_HOME=str(ROOT / 'tools/jdk-25.0.5+8'),
    GHIDRA_INSTALL_DIR=str(ROOT / 'tools/ghidra_12.2_PUBLIC'),
    XDG_CONFIG_HOME=str(OUT / 'config'),
    XDG_CACHE_HOME=str(OUT / 'cache'),
    XDG_DATA_HOME=str(OUT / 'data'),
    TMPDIR=str(OUT / 'tmp'),
    JAVA_TOOL_OPTIONS=f'-Duser.home={OUT / "home"} -Djava.io.tmpdir={OUT / "tmp"}',
)
env['PATH'] = str(ROOT / 'tools/jdk-25.0.5+8/bin') + os.pathsep + env['PATH']
args = [
    '--headless', '--binary', str(BINARY), '--project', str(PROJECT),
    '--project-name', 'independent-endpoint-01', '--no-analyze', '--discard',
    '--language', 'PowerPC:BE:32:default', '--compiler', 'default',
    '--extension-jar', str(JAR), '--ready-timeout', '540',
]
report = {
    'started_at_utc': datetime.now(timezone.utc).isoformat(),
    'status': 'starting', 'binary_sha256': DATA_SHA, 'jar_sha256': JAR_SHA,
    'script_sha256': hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
    'command': str(ROOT / 'tools/ghidramcp/.venv/bin/ghidra-mcp'),
    'args': args, 'project_scope': 'saved independent server project, --no-analyze --discard',
    'queries': [],
}


def save():
    (OUT / 'report.json').write_text(json.dumps(report, indent=2) + '\n')


async def main():
    params = StdioServerParameters(command=report['command'], args=args, env=env, cwd=str(ROOT))
    try:
        with (OUT / 'server.stderr.log').open('w') as err:
            async with stdio_client(params, errlog=err) as (reader, writer), ClientSession(
                reader, writer, read_timeout_seconds=timedelta(seconds=540)
            ) as session:
                init = await session.initialize()

                async def query(label, tool, arguments):
                    result = await session.call_tool(tool, arguments, read_timeout_seconds=timedelta(seconds=300))
                    payload = {
                        'captured_at_utc': datetime.now(timezone.utc).isoformat(),
                        'query': {'id': label, 'tool': tool, 'arguments': arguments},
                        'result': result.model_dump(mode='json'),
                    }
                    (OUT / 'raw' / f'{label}.json').write_text(json.dumps(payload, indent=2) + '\n')
                    report['queries'].append({'id': label, 'tool': tool, 'arguments': arguments, 'is_error': bool(result.isError)})
                    save()
                    print(label, 'ERROR' if result.isError else 'OK', flush=True)
                    return result.structuredContent

                before = await query('identity-before', 'get_program_info', {})
                assert before and before['sha256'] == DATA_SHA
                assert before['language_id'] == 'PowerPC:BE:32:default'
                assert before['function_count'] == 1840 and before['min_address'] == '10000000'
                report['identity_before'] = before
                report['mcp_server'] = init.serverInfo.model_dump(mode='json')
                report['status'] = 'querying'
                save()
                result = await query('disassemble-10015500-tail', 'disassemble_function', {'address': '10015500', 'offset': 500, 'limit': 100})
                assert result and 'items' in result and len(result['items']) == 55
                assert result.get('total') == 555 and result.get('has_more') is False
                after = await query('identity-after', 'get_program_info', {})
                assert after == before
                report['identity_after'] = after
        report['status'] = 'complete'
    except BaseException as exc:
        report['status'] = 'failed-preserved'
        report['error'] = f'{type(exc).__name__}: {exc}'
        raise
    finally:
        report['finished_at_utc'] = datetime.now(timezone.utc).isoformat()
        save()


save()
asyncio.run(main())
