def digest(data):

parity = json.loads((source / "offline-parity.json").read_text())["cases"]["server-name-162"]
routes_by_file = {}
for kind, routes in parity["routes"].items():
    for route in routes:
        routes_by_file.setdefault(route[0], []).append((kind, route))
selection = {}
for filename, routes in routes_by_file.items():
    packets = read_packets(source / filename)
    indexes = set()
    observations = []
    for kind, (_, src_ip, src_port, dst_ip, dst_port) in routes:
        selected = [p for p in packets if
                    ((p["src_ip"], p["src_port"], p["dst_ip"], p["dst_port"]) == (src_ip, src_port, dst_ip, dst_port)
                    or (p["src_ip"], p["src_port"], p["dst_ip"], p["dst_port"]) == (dst_ip, dst_port, src_ip, src_port))]
        syns = [p for p in selected if (p["src_ip"], p["src_port"]) == (src_ip, src_port) and p["flags"] & 2 and not p["flags"] & 16]
        assert len(syns) == 1, (filename, kind, "SYN")
        assert all(p["timestamp_us"] >= syns[0]["timestamp_us"] for p in selected)
        indexes.update(p["index"] for p in selected)
        observations.append({"kind": kind, "route": [src_ip, src_port, dst_ip, dst_port],
                             "syn_index": syns[0]["index"], "syn_timestamp_us": syns[0]["timestamp_us"],
                             "packet_count": len(selected), "terminal_flags": [p["flags"] for p in selected if p["flags"] & 5]})
    raw = (source / filename).read_bytes()
    assert raw[:4] == bytes.fromhex("d4c3b2a1")
    output = bytearray(raw[:24]); offset = 24; index = 0
    while offset < len(raw):
        index += 1
        assert offset + 16 <= len(raw)
        _, _, size, original = struct.unpack_from("<IIII", raw, offset)
        assert size == original and offset + 16 + size <= len(raw)
        if index in indexes:
            output += raw[offset:offset + 16 + size]
        offset += 16 + size
    assert offset == len(raw)
    target = dest / "pcaps" / filename
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_bytes(output)
    assert len(read_packets(target)) == len(indexes), filename
    selection[filename] = {"original_path": str((source / filename).relative_to(mothra)),
                           "original_bytes": len(raw), "original_sha256": digest(raw),
                           "selected_bytes": len(output), "selected_sha256": digest(output),
                           "selected_record_indexes": sorted(indexes), "routes": observations}
