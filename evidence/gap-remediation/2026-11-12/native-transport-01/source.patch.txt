+/* TCP/IP overlays keep 32-bit queue links, including the control-block
+ * sentinel and packet/mbuf pointers. Linux thread arenas need not be below
+ * 4 GiB. Keep only these linked allocations in a representable mapping. */
+#if defined(__linux__) && defined(__x86_64__)
+#include <sys/mman.h>
+#include <stdint.h>
+#include <errno.h>
+
+union slirp_low_header {
+    size_t size;
+    long double alignment;
+    void *pointer_alignment;
+};
+
+void *slirp_low_alloc(size_t size)
+{
+    union slirp_low_header *header;
+    if (size > UINT32_MAX - sizeof(*header)) {
+        errno = ENOMEM;
+        return NULL;
+    }
+    header = (union slirp_low_header *)mmap(NULL, sizeof(*header) + size,
+        PROT_READ | PROT_WRITE, MAP_PRIVATE | MAP_ANONYMOUS | MAP_32BIT, -1, 0);
+    if (header == MAP_FAILED)
+        return NULL;
+    header->size = size;
+    return header + 1;
+}
+
+void slirp_low_free(void *pointer)
+{
+    if (pointer) {
+        union slirp_low_header *header = (union slirp_low_header *)pointer - 1;
+        munmap(header, sizeof(*header) + header->size);
+    }
+}
+
+void *slirp_low_realloc(void *pointer, size_t size)
+{
+    void *replacement;
+    size_t previous;
+    if (!pointer)
+        return slirp_low_alloc(size);
+    if (!size) {
+        slirp_low_free(pointer);
+        return NULL;
+    }
+    previous = ((union slirp_low_header *)pointer - 1)->size;
+    replacement = slirp_low_alloc(size);
+    if (!replacement)
+        return NULL;
+    memcpy(replacement, pointer, previous < size ? previous : size);
+    slirp_low_free(pointer);
+    return replacement;
+}
+#else
+void *slirp_low_alloc(size_t size) { return malloc(size); }
+void slirp_low_free(void *pointer) { free(pointer); }
+void *slirp_low_realloc(void *pointer, size_t size) { return realloc(pointer, size); }
+#endif
+/* MOTHRA_LOW_MEMORY_END */
